--------------------------------------------------------------------------------
Backend: REST API with Express and MongoDB Mongoose
File: backend/server.js
--------------------------------------------------------------------------------
const express = require("express");
const mongoose = require("mongoose");
const cors = require("cors");

const app = express();
app.use(cors());
app.use(express.json());

// MongoDB connection
mongoose.connect("mongodb://127.0.0.1:27017/MERNDB");

// Schema & Model
const StudentSchema = new mongoose.Schema({
  name: String
});

const Student = mongoose.model("Student", StudentSchema);

// GET: Retrieve all students
app.get("/students", async (req, res) => {
  const students = await Student.find();
  res.json(students);
});

// POST: Save a new student
app.post("/students", async (req, res) => {
  const student = new Student({
    name: req.body.name
  });
  await student.save();
  res.json({
    message: "Student Saved"
  });
});

app.listen(5000, () => {
  console.log("Server running on port 5000");
});

--------------------------------------------------------------------------------
Frontend: React Application with Axios Integration
File: frontend/src/App.jsx
--------------------------------------------------------------------------------
import { useState, useEffect } from "react";
import axios from "axios";

function App() {
  const [students, setStudents] = useState([]);
  const [name, setName] = useState("");

  const fetchStudents = () => {
    axios.get("http://localhost:5000/students")
      .then(response => {
        setStudents(response.data);
      });
  };

  useEffect(() => {
    fetchStudents();
  }, []);

  const submitData = () => {
    axios.post("http://localhost:5000/students", { name: name })
      .then(() => {
        setName("");
        fetchStudents();
      });
  };

  return (
    <div>
      <h1>Student List</h1>
      <input
        type="text"
        placeholder="Enter Name"
        value={name}
        onChange={(e) => setName(e.target.value)}
      />
      <button onClick={submitData}>Submit</button>

      <h2>Stored Students</h2>
      <ul>
        {students.map(student => (
          <li key={student._id}>{student.name}</li>
        ))}
      </ul>
    </div>
  );
}

export default App;
