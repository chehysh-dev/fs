C. arrowFunction.js - Arrow Functions
--------------------------------------------------------------------------------
const square = (num) => num * num;

const multiply = (a, b) => {
    return a * b;
};

console.log("Square =", square(5));
console.log("Multiplication =", multiply(4, 6));

--------------------------------------------------------------------------------
D. arrayMethods.js - Array Methods (map, filter, reduce)
--------------------------------------------------------------------------------
let numbers = [1, 2, 3, 4, 5];

let squares = numbers.map(num => num * num);
let evenNumbers = numbers.filter(num => num % 2 === 0);
let total = numbers.reduce((sum, num) => sum + num, 0);

console.log("Original Array:", numbers);
console.log("Squares:", squares);
console.log("Even Numbers:", evenNumbers);
console.log("Sum:", total);

F. studentManagement.js - Functions, Arrays, and Objects Combined
--------------------------------------------------------------------------------
let students = [
    { id: 1, name: "Rahul", marks: 80 },
    { id: 2, name: "Priya", marks: 90 },
    { id: 3, name: "Amit", marks: 70 }
];

function displayStudents(studentArray) {
    studentArray.forEach(student => {
        console.log(
            `ID: ${student.id}, Name: ${student.name}, Marks: ${student.marks}`
        );
    });
}

function calculateAverage(studentArray) {
    let totalMarks = studentArray.reduce(
        (sum, student) => sum + student.marks,
        0
    );
    return totalMarks / studentArray.length;
}

console.log("Student Details:");
displayStudents(students);
console.log("Average Marks =", calculateAverage(students));
