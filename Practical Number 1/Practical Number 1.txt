3. control.js - Control Statements (if-else & switch-case)
--------------------------------------------------------------------------------
let age = 18;
if (age >= 18) {
    console.log("eligible to vote");
} else {
    console.log("not eligible to vote");
}

let marks = 75;
if (marks >= 80) {
    console.log("grade A ");
} else if (marks >= 60) {
    console.log("grade B");
} else {
    console.log("grade C");
}

let day = 3;
switch (day) {
    case 1:
        console.log("monday");
        break;
    case 2:
        console.log("tuesday");
        break;
    case 3:
        console.log("wednesday");
        break;
    default:
        console.log("invaild day");
}
