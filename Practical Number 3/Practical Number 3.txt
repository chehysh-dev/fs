C. Declare Variables and Write a Function to Calculate the Sum of Two Numbers
File: App.jsx
--------------------------------------------------------------------------------
function App() {
  const num1 = 20;
  const num2 = 30;

  function calculateSum(a, b) {
    return a + b;
  }

  return (
    <div>
      <h1>Addition Program</h1>
      <p>Number 1: {num1}</p>
      <p>Number 2: {num2}</p>
      <p>Sum: {calculateSum(num1, num2)}</p>
    </div>
  );
}

export default App;

F. Create a React Component that Displays the Current Date and Time Dynamically
Files: DateTime.jsx & App.jsx
--------------------------------------------------------------------------------
// DateTime.jsx
function DateTime() {
  const currentDate = new Date();

  return (
    <div>
      <h2>Current Date and Time</h2>
      <p>Date: {currentDate.toLocaleDateString()}</p>
      <p>Time: {currentDate.toLocaleTimeString()}</p>
    </div>
  );
}

export default DateTime;

// App.jsx
import DateTime from './DateTime';

function App() {
  return (
    <div>
      <h1>React Date and Time</h1>
      <DateTime />
    </div>
  );
}

export default App;
